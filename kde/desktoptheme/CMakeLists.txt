install(DIRECTORY .
        DESTINATION ${KDE_INSTALL_DATADIR}/plasma/desktoptheme/BlossomUI
        PATTERN "CMakeLists.txt" EXCLUDE
        PATTERN "metadata.json" EXCLUDE)

file(GLOB_RECURSE BLOSSOM_DESKTOPTHEME_FILES CONFIGURE_DEPENDS
     LIST_DIRECTORIES false
     RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}
     ${CMAKE_CURRENT_SOURCE_DIR}/*)
list(SORT BLOSSOM_DESKTOPTHEME_FILES)
set(BLOSSOM_DESKTOPTHEME_CONTENT "")
foreach(theme_file IN LISTS BLOSSOM_DESKTOPTHEME_FILES)
    if(NOT theme_file STREQUAL "CMakeLists.txt")
        file(SHA1 ${CMAKE_CURRENT_SOURCE_DIR}/${theme_file} theme_file_hash)
        string(APPEND BLOSSOM_DESKTOPTHEME_CONTENT "${theme_file}:${theme_file_hash};")
    endif()
endforeach()
string(SHA1 BLOSSOM_DESKTOPTHEME_HASH "${BLOSSOM_DESKTOPTHEME_CONTENT}")
string(SUBSTRING ${BLOSSOM_DESKTOPTHEME_HASH} 0 8 BLOSSOM_DESKTOPTHEME_HASH)

file(READ ${CMAKE_CURRENT_SOURCE_DIR}/metadata.json BLOSSOM_DESKTOPTHEME_METADATA)
string(REGEX REPLACE "\"Version\": \"[^\"]*\""
       "\"Version\": \"${PROJECT_VERSION}+${BLOSSOM_DESKTOPTHEME_HASH}\""
       BLOSSOM_DESKTOPTHEME_METADATA "${BLOSSOM_DESKTOPTHEME_METADATA}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/metadata.json "${BLOSSOM_DESKTOPTHEME_METADATA}")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/metadata.json
        DESTINATION ${KDE_INSTALL_DATADIR}/plasma/desktoptheme/BlossomUI)
